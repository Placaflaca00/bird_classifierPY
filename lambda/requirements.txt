# Runtime dependencies para la imagen AWS Lambda (Python 3.12 / AL2023).
# Mantener al mínimo: cada MB inflado = cold-start más lento.
# Versiones pinneadas para builds reproducibles.

# Runtime de inferencia BirdNET TFLite (LiteRT, GA desde TF 2.21, reemplaza tflite-runtime)
# Requiere manylinux_2_27+ -> AL2023 obligatorio (glibc 2.34).
ai-edge-litert==2.1.4

# Runtime de inferencia ONNX
onnxruntime==1.26.0

# Audio I/O: soundfile 0.13.1 trae libsndfile 1.1+ bundleado (con soporte MP3 nativo).
# librosa solo se usa para load + resample a 48 kHz.
librosa==0.11.0
soundfile==0.13.1

# Bound de numpy: onnxruntime 1.26 soporta numpy 2.x.
numpy>=2.0,<3

# AWS SDK (Fase 4): pinned >=1.31.55 para ReturnValuesOnConditionCheckFailure
# (usado en _check_and_increment_rate_limit para devolver requests_today en
# rejects sin GetItem extra). Pin exacto a versión del venv para parity
# tests-local <-> Docker. boto3 ya viene en el base image AWS Lambda Python
# 3.12 pero NO garantiza versión — pip upgradea al instalar este.
boto3==1.43.9
# Bound a versión exacta de la imagen Docker (1.43.9 < 1.43.10 = drift de patch).
# Aunque drift de patch es bajo riesgo, pin exacto mantiene parity completa
# tests-local <-> container <-> Lambda runtime.
botocore==1.43.9
